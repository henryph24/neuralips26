% Standalone TikZ: Normalization-Induced Routing Collapse Mechanism
% Usage: \input{figures/tikz_collapse_mechanism.tex} inside a figure environment
%
% Academic-style two-panel figure modeled on the gemini-academic-viz reference
% (figures/talk/collapse_mechanism_v2.png). Conventions:
%   * white-fill panels with thin colored strokes (no solid color fills)
%   * desaturated muted palette (dusty blue / sage / brick / ochre)
%   * single dominant stroke weight (~0.7pt); only highlighted E1 thicker
%   * thin line arrows with small triangular arrowheads (no block arrows)
%   * plain black caption text (no colored captions)
%   * lowercase bold (a) / (b) panel labels per NeurIPS convention
%
% Routing pattern: all routing arrows emerge BELOW the panel stack and fan
% out to the experts. Color encoding (blue / green / red / ochre) carries the
% waveform-to-expert identity, so no arrows need to cross or pass through
% any panel.

\definecolor{cmDustyBlue}{RGB}{74,111,165}
\definecolor{cmSageGreen}{RGB}{107,142,90}
\definecolor{cmBrickRed}{RGB}{168,74,74}
\definecolor{cmOchre}{RGB}{184,137,62}
\definecolor{cmGhost}{RGB}{176,176,176}

\begin{tikzpicture}[
    wavepanel/.style={rectangle, draw=#1, fill=white, line width=0.7pt,
        rounded corners=1.5pt, minimum width=4cm, minimum height=0.65cm,
        inner sep=2pt},
    wavepanelgray/.style={rectangle, draw=cmGhost, fill=white, line width=0.6pt,
        rounded corners=1.5pt, minimum width=4cm, minimum height=0.65cm,
        inner sep=2pt},
    expert/.style={rectangle, draw=#1, fill=white, line width=0.7pt,
        rounded corners=2pt, minimum width=0.85cm, minimum height=0.5cm,
        font=\scriptsize\bfseries, text=#1},
    expertactive/.style={rectangle, draw=cmBrickRed, fill=white, line width=1.0pt,
        rounded corners=2pt, minimum width=0.85cm, minimum height=0.5cm,
        font=\scriptsize\bfseries, text=cmBrickRed},
    expertghost/.style={rectangle, draw=cmGhost, fill=white, line width=0.5pt,
        rounded corners=2pt, minimum width=0.85cm, minimum height=0.5cm,
        font=\scriptsize, text=cmGhost},
    arr/.style={-{Stealth[length=1.7mm,width=1.3mm]}, line width=0.65pt, #1},
    normarr/.style={-{Stealth[length=2mm,width=1.5mm]}, line width=0.7pt, black!75},
    panellabel/.style={font=\small\bfseries, anchor=west, text=black},
    caption/.style={font=\small, anchor=north, text=black},
]

% ============================================================
% (a) Raw input space
% ============================================================
\node[panellabel] at (0.0, 5.10) {(a) Raw input space};

% Four stacked waveform panels (centered at x=2)
\node[wavepanel=cmDustyBlue]  (p1) at (2.0, 4.40) {};
\node[wavepanel=cmSageGreen]  (p2) at (2.0, 3.65) {};
\node[wavepanel=cmBrickRed]   (p3) at (2.0, 2.90) {};
\node[wavepanel=cmOchre]      (p4) at (2.0, 2.15) {};

% Waveforms inside panels
\draw[cmDustyBlue, line width=0.6pt]
    plot[domain=0.20:3.80, samples=110, smooth]
    ({\x},{4.40+0.20*sin(deg(7.5*(\x-2.0)))});
\draw[cmSageGreen, line width=0.6pt]
    plot[domain=0.20:3.80, samples=80, smooth]
    ({\x},{3.65+0.22*tanh(1.6*(\x-2.0))});
\draw[cmBrickRed, line width=0.6pt]
    (0.25, 2.74) -- (3.75, 3.06);
\draw[cmOchre, line width=0.55pt]
    plot[domain=0.20:3.80, samples=130]
    ({\x},{2.15+0.16*rand});

% Experts in a horizontal row below (K=5 to match the paper).
% Spread across [0.20, 3.80] so the routing arrows fan out from beneath the
% panel stack without crossing each other.
\node[expert=cmDustyBlue] (e1) at (0.30, 1.20) {E1};
\node[expert=cmSageGreen] (e2) at (1.20, 1.20) {E2};
\node[expert=cmBrickRed]  (e3) at (2.10, 1.20) {E3};
\node[expert=cmOchre]     (e4) at (3.00, 1.20) {E4};
\node[expert=cmGhost]     (e5) at (3.90, 1.20) {E5};

% Routing arrows: emerge from below the bottom panel (p4) at five different
% x positions, each going down to its color-matched expert.
% Color carries the waveform-to-expert identity. E5 stays unused on this
% sample window (gray) — the diagram shows that *different* windows light up
% different experts, not that every expert fires every window.
\draw[arr=cmDustyBlue] (0.30, 1.83) -- (e1.north);
\draw[arr=cmSageGreen] (1.20, 1.83) -- (e2.north);
\draw[arr=cmBrickRed]  (2.10, 1.83) -- (e3.north);
\draw[arr=cmOchre]     (3.00, 1.83) -- (e4.north);

\node[caption] at (2.0, 0.60) {Diverse routing};

% ============================================================
% Middle: instance normalization box
% ============================================================
\node[rectangle, draw=black!75, fill=white, rounded corners=2pt,
      line width=0.7pt, font=\scriptsize, align=center, inner sep=4pt]
      (norm) at (5.85, 3.30)
      {Instance normalization\\strips $\mu, \sigma$};

\draw[normarr] (4.10, 3.30) -- (norm.west);
\draw[normarr] (norm.east) -- (7.60, 3.30);

% ============================================================
% (b) Normalized space
% ============================================================
\node[panellabel] at (7.80, 5.10) {(b) Normalized space};

% Four IDENTICAL light-gray panels — central message
\node[wavepanelgray] (n1) at (10.0, 4.40) {};
\node[wavepanelgray] (n2) at (10.0, 3.65) {};
\node[wavepanelgray] (n3) at (10.0, 2.90) {};
\node[wavepanelgray] (n4) at (10.0, 2.15) {};

\foreach \y in {4.40, 3.65, 2.90, 2.15} {
    \draw[cmGhost, line width=0.55pt]
        plot[domain=8.20:11.80, samples=80, smooth]
        ({\x},{\y+0.10*sin(deg(3.0*(\x-10.0)))});
}

% Experts: only E1 highlighted, E2-E5 ghosted (K=5 matches the paper).
\node[expertactive] (re1) at (8.20, 1.20) {E1};
\node[expertghost]  (re2) at (9.10, 1.20) {E2};
\node[expertghost]  (re3) at (10.00, 1.20) {E3};
\node[expertghost]  (re4) at (10.90, 1.20) {E4};
\node[expertghost]  (re5) at (11.80, 1.20) {E5};

% Convergent ghost arrows: five arrows below the panel stack, all pointing
% to E1 — the visual signature of routing collapse.
\draw[arr=cmGhost] (8.20, 1.83) -- (re1.north);
\draw[arr=cmGhost] (9.10, 1.83) -- ([xshift=0.07cm]re1.north);
\draw[arr=cmGhost] (10.00, 1.83) -- ([xshift=0.14cm]re1.north);
\draw[arr=cmGhost] (10.90, 1.83) -- ([xshift=0.21cm]re1.north);
\draw[arr=cmGhost] (11.80, 1.83) -- ([xshift=0.28cm]re1.north);

\node[caption] at (10.0, 0.60) {Collapse: $H = 0.000$};

\end{tikzpicture}
